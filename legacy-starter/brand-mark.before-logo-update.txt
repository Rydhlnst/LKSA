import Link from "next/link";

type BrandMarkProps = { settings: { organizationName: string; shortName: string; logoPrimary: string; logoSecondary: string } };

function LogoFallback({ label, tone }: { label: string; tone: "orange" | "green" }) {
  return <span className={`brand-mark brand-mark-${tone}`} aria-hidden="true">{label.slice(0, 1)}</span>;
}

export function BrandMark({ settings }: BrandMarkProps) {
  return (
    <Link href="/" className="flex min-w-0 items-center gap-3" aria-label={`${settings.organizationName} home`}>
      {settings.logoPrimary ? <img src={settings.logoPrimary} alt="Logo utama" className="h-10 w-10 rounded-full object-cover" /> : <LogoFallback label="M" tone="orange" />}
      <span className="hidden min-w-0 lg:block">
        <span className="block truncate font-heading text-base font-bold leading-tight text-ink">{settings.organizationName}</span>
        <span className="block text-[10px] font-semibold uppercase tracking-[0.2em] text-muted">Children Center</span>
      </span>
      {settings.logoSecondary ? <img src={settings.logoSecondary} alt="Logo organisasi" className="hidden h-10 w-10 rounded-full object-cover sm:block" /> : <LogoFallback label="P" tone="green" />}
    </Link>
  );
}
