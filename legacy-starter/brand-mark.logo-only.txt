import Image from "next/image";
import Link from "next/link";

type BrandMarkProps = { settings: { organizationName: string; shortName: string; logoPrimary: string; logoSecondary: string } };

function LogoFallback({ label, tone }: { label: string; tone: "orange" | "green" }) {
  return <span className={`brand-mark brand-mark-${tone}`} aria-hidden="true">{label.slice(0, 1)}</span>;
}

export function BrandMark({ settings }: BrandMarkProps) {
  return (
    <Link href="/" className="flex min-w-0 items-center gap-3" aria-label={`${settings.organizationName} home`}>
      {settings.logoPrimary ? <Image src={settings.logoPrimary} alt={`Logo ${settings.shortName}`} width={44} height={44} className="h-11 w-11 rounded-full object-cover" priority /> : <LogoFallback label="M" tone="orange" />}
      <span className="min-w-0 lg:block">
        <span className="block max-w-[220px] truncate font-heading text-base font-bold leading-tight text-ink sm:max-w-[280px]">{settings.organizationName}</span>
        <span className="block text-[10px] font-semibold uppercase tracking-[0.2em] text-muted">Children Center</span>
      </span>
      {settings.logoSecondary ? <Image src={settings.logoSecondary} alt="Logo organisasi" width={44} height={44} className="hidden h-11 w-11 rounded-full object-cover sm:block" /> : <LogoFallback label="P" tone="green" />}
    </Link>
  );
}
